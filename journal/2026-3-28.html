        if (path.startsWith("/journal/2026-")) {

            if (!content.startsWith("<h1>Error Fetching Content</h1>")) {
                request->send(200, "text/html", content);
                return;
            }
        }
        AsyncWebServerResponse *response = request->beginResponse(301);
        response->addHeader("Location", "/journal/");
        response->addHeader("Cache-Control", "max-age=86400");
        request->send(response);
        return;
    }

    request->send(404, "text/html", fetch("/404.html"));
});
